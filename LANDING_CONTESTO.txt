  .modal                  ← contenuto modale
  .triage-loading         ← spinner durante demo (id="XXX-loading")
  .demo-output            ← area output demo (id="XXX-output")
  .out-row                ← riga chiave/valore nell'output demo
  .out-label              ← etichetta sopra l'output

--------------------------------------------------------------------------------
ARCHITETTURA TECH
--------------------------------------------------------------------------------

- Single file HTML (tutto inline: CSS + JS)
- Zero librerie JS esterne
- Zero backend: chiamate API demo sono simulate (setTimeout 2000ms)
- Modali: openModal(id) / closeModal(id) già presenti nel file
- Demo simulate: ogni tool ha runXXX() con dati pre-scritti
- Dark mode: non implementata in landing (solo in index.html e nei tool)
- Analytics: Umami (script asincrono in <head>)

TEMPLATE — FUNZIONE JS DEMO
--------------------------------------------------------------------------------

Inserire dentro il blocco <script>, prima della funzione runBriefing().
Sostituire XXX con l'id del tool.

function runXXX(){
  const scenario=document.getElementById('XXX-scenario').value;
  const btn=document.getElementById('XXX-btn');
  const loading=document.getElementById('XXX-loading');
  const output=document.getElementById('XXX-output');
  btn.disabled=true;btn.style.opacity='.5';
  output.classList.remove('visible');
  loading.classList.add('visible');
  const risultati={
    scenario1:{
      campo1:'Valore 1',
      campo2:'Valore 2',
      testo:'Testo descrittivo del risultato scenario 1.'
    },
    scenario2:{
      campo1:'Valore A',
      campo2:'Valore B',
      testo:'Testo descrittivo del risultato scenario 2.'
    },
    scenario3:{
      campo1:'Valore X',
      campo2:'Valore Y',
      testo:'Testo descrittivo del risultato scenario 3.'
    }
  };
  setTimeout(()=>{
    loading.classList.remove('visible');
    btn.disabled=false;btn.style.opacity='1';
    const r=risultati[scenario];
    document.getElementById('XXX-result').innerHTML=`
      <div class="out-row"><span class="key">Campo 1</span><span class="val">${r.campo1}</span></div>
      <div class="out-row"><span class="key">Campo 2</span><span class="val">${r.campo2}</span></div>
      <p style="margin-top:14px;font-size:.88rem;color:#555;line-height:1.6;">${r.testo}</p>
      <p style="margin-top:8px;font-size:.75rem;color:#888;">Dataset dimostrativo — dati fittizi.</p>`;
    output.classList.add('visible');
  },2000);
}
